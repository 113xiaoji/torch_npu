cmake_minimum_required(VERSION 3.16)
project(eager_prepare_core_tests LANGUAGES CXX)
find_package(Threads REQUIRED)
add_executable(eager_prepare_core_tests eager_prepare_core_test.cpp)
target_compile_features(eager_prepare_core_tests PRIVATE cxx_std_17)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
target_include_directories(eager_prepare_core_tests PRIVATE "${REPO_ROOT}")
target_link_libraries(eager_prepare_core_tests PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(eager_prepare_core_tests PRIVATE /W4 /WX /EHsc)
else()
  target_compile_options(eager_prepare_core_tests PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_test(NAME eager_prepare_core COMMAND eager_prepare_core_tests)
set_tests_properties(eager_prepare_core PROPERTIES TIMEOUT 45)
