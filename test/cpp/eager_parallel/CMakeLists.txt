cmake_minimum_required(VERSION 3.16)
project(r26_eager_parallel_core_tests LANGUAGES CXX)

# Standalone: no Torch, Python, CANN or parent PTA configure is needed.
option(R26_ENABLE_ASAN "Instrument native protocol tests with AddressSanitizer" OFF)
option(R26_ENABLE_TSAN "Instrument native protocol tests with ThreadSanitizer" OFF)
if(R26_ENABLE_ASAN AND R26_ENABLE_TSAN)
  message(FATAL_ERROR "ASan and TSan require separate build directories")
endif()

find_package(Threads REQUIRED)
add_executable(r26_dependency_scheduler_test dependency_scheduler_test.cpp)
add_executable(r26_feature_pipeline_test feature_pipeline_test.cpp)
add_executable(r27_immediate_protocol_test immediate_protocol_test.cpp)
set(R26_TEST_TARGETS r26_dependency_scheduler_test r26_feature_pipeline_test r27_immediate_protocol_test)
foreach(R26_TARGET IN LISTS R26_TEST_TARGETS)
  target_compile_features(${R26_TARGET} PRIVATE cxx_std_17)
  target_include_directories(${R26_TARGET} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
  target_link_libraries(${R26_TARGET} PRIVATE Threads::Threads)
  if(MSVC)
    target_compile_options(${R26_TARGET} PRIVATE /W4 /WX /EHsc)
  else()
    target_compile_options(${R26_TARGET} PRIVATE -Wall -Wextra -Werror)
  endif()
endforeach()
if(R26_ENABLE_ASAN OR R26_ENABLE_TSAN)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" OR WIN32)
    message(FATAL_ERROR "R26 sanitizer presets require a supported Linux GCC/Clang toolchain")
  endif()
  if(R26_ENABLE_ASAN)
    set(R26_SANITIZER address)
  else()
    set(R26_SANITIZER thread)
  endif()
  foreach(R26_TARGET IN LISTS R26_TEST_TARGETS)
    target_compile_options(${R26_TARGET} PRIVATE
      -fsanitize=${R26_SANITIZER} -fno-omit-frame-pointer -g)
    target_link_options(${R26_TARGET} PRIVATE -fsanitize=${R26_SANITIZER})
  endforeach()
endif()

enable_testing()
add_test(NAME r26_dependency_scheduler COMMAND r26_dependency_scheduler_test)
add_test(NAME r26_feature_pipeline COMMAND r26_feature_pipeline_test)
add_test(NAME r27_immediate_protocol COMMAND r27_immediate_protocol_test)
set_tests_properties(r26_dependency_scheduler r26_feature_pipeline r27_immediate_protocol PROPERTIES TIMEOUT 20)
